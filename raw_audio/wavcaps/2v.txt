import os
import glob
import json
import zipfile
import io
import torch
import torchaudio
import torchaudio.transforms as T
import laion_clap
import wget
import numpy as np
from tqdm import tqdm

# --- КОНФИГУРАЦИЯ ---
# Пути к исходным данным (поменяй на свои)
WAVCAPS_ZIP_FOLDER = "data/raw/wavcaps/zips"   # Папка где лежат .zip файлы
WAVCAPS_JSON = "data/raw/wavcaps/wavcaps.json" # JSON с описаниями

# Пути к MusicCaps (если есть)
MUSICCAPS_FOLDER = "data/raw/musiccaps" 
MUSICCAPS_CSV = "data/raw/musiccaps/musiccaps.csv"

# Куда сохранять результат
OUTPUT_DIR = "data/processed"
CHECKPOINT_NAME = "music_audioset_epoch_15_esc_90.14.pt"

# Параметры аудио
TARGET_SR = 48000     # CLAP требует 48kHz
MAX_SECONDS = 10      # Максимальная длина аудио
MAX_LENGTH = TARGET_SR * MAX_SECONDS
BATCH_SIZE = 64
DEVICE = "cuda" if torch.cuda.is_available() else "cpu"

def download_weights_if_needed():
    """Скачивает правильный чекпоинт для HTSAT-base"""
    url = "https://huggingface.co/lukewys/laion_clap/resolve/main/music_audioset_epoch_15_esc_90.14.pt"
    if not os.path.exists(CHECKPOINT_NAME):
        print(f"⬇️ Скачиваем веса {CHECKPOINT_NAME}...")
        wget.download(url, CHECKPOINT_NAME)
        print("\n✅ Скачано.")
    return CHECKPOINT_NAME

def load_clap_model():
    print(f"🔊 Loading CLAP model on {DEVICE}...")
    ckpt_path = download_weights_if_needed()
    
    # ВАЖНО: amodel='HTSAT-base' соответствует скачанным весам
    model = laion_clap.CLAP_Module(enable_fusion=False, amodel='HTSAT-base')
    
    # Загружаем веса явно
    model.load_ckpt(ckpt_path)
    
    model.to(DEVICE)
    model.eval()
    return model

def process_audio_tensor(waveform, sr):
    """Приводит аудио к единому формату (48k, mono, 10sec)"""
    # 1. Resample
    if sr != TARGET_SR:
        resampler = T.Resample(sr, TARGET_SR)
        waveform = resampler(waveform)
    
    # 2. Mono
    if waveform.shape[0] > 1:
        waveform = waveform.mean(dim=0, keepdim=True)
        
    # 3. Pad or Trim (Фиксированная длина)
    # Это критично для батчинга!
    if waveform.shape[1] > MAX_LENGTH:
        waveform = waveform[:, :MAX_LENGTH]
    elif waveform.shape[1] < MAX_LENGTH:
        pad_size = MAX_LENGTH - waveform.shape[1]
        waveform = torch.nn.functional.pad(waveform, (0, pad_size))
    
    # Возвращаем [Length] (без канала)
    return waveform.squeeze(0)

def get_wavcaps_mapping(json_path):
    print("📖 Reading WavCaps JSON...")
    if not os.path.exists(json_path):
        print(f"⚠️ JSON not found: {json_path}")
        return {}
        
    with open(json_path, 'r') as f:
        data = json.load(f)
    
    mapping = {}
    # Адаптация под структуру WavCaps (может отличаться)
    items = data['data'] if 'data' in data else data
    
    for item in items:
        # ID обычно совпадает с именем файла
        key = str(item['id']) 
        caption = item['caption']
        mapping[key] = caption
    return mapping

def cache_datasets():
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    
    # 1. Загрузка модели
    try:
        model = load_clap_model()
    except Exception as e:
        print(f"❌ Critical Error loading model: {e}")
        return

    all_embeddings = []
    all_captions = []
    
    # Буферы
    batch_audio = []
    batch_caps = []

    def flush_batch():
        nonlocal batch_audio, batch_caps
        if not batch_audio:
            return
            
        try:
            with torch.no_grad():
                # Стек тензоров [Batch, Length]
                batch_tensor = torch.stack(batch_audio).to(DEVICE) # Отправляем на GPU
                
                # Инференс
                # use_tensor=True значит мы подаем готовые тензоры, а не пути к файлам
                embeddings = model.get_audio_embedding_from_data(x=batch_tensor, use_tensor=True)
                
                # Сохраняем на CPU
                all_embeddings.append(embeddings.cpu())
                all_captions.extend(batch_caps)
        except Exception as e:
            print(f"⚠️ Batch processing error: {e}")
        finally:
            batch_audio = []
            batch_caps = []

    # --- WAVCAPS LOOP ---
    wavcaps_map = get_wavcaps_mapping(WAVCAPS_JSON)
    zip_files = glob.glob(os.path.join(WAVCAPS_ZIP_FOLDER, "*.zip"))
    
    print(f"📦 Processing {len(zip_files)} zip archives...")
    
    for zip_path in tqdm(zip_files, desc="Archives"):
        try:
            with zipfile.ZipFile(zip_path, 'r') as z:
                # Фильтруем только аудио
                files = [f for f in z.namelist() if f.endswith(('.flac', '.wav'))]
                
                for filename in files:
                    # Ищем ID (обычно имя файла без расширения)
                    file_id = os.path.splitext(os.path.basename(filename))[0]
                    
                    if file_id not in wavcaps_map:
                        continue
                        
                    try:
                        # Читаем в RAM
                        audio_bytes = z.read(filename)
                        wav, sr = torchaudio.load(io.BytesIO(audio_bytes))
                        
                        # Обрабатываем
                        processed = process_audio_tensor(wav, sr)
                        
                        batch_audio.append(processed)
                        batch_caps.append(wavcaps_map[file_id])
                        
                        if len(batch_audio) >= BATCH_SIZE:
                            flush_batch()
                            
                    except Exception as e:
                        # Битое аудио пропускаем
                        continue
                        
        except Exception as e:
            print(f"❌ Error reading zip {zip_path}: {e}")

    # Обработка хвоста
    flush_batch()
    
    # --- СОХРАНЕНИЕ ---
    if not all_embeddings:
        print("❌ No data processed. Check paths!")
        return

    print("💾 Saving artifacts...")
    final_tensor = torch.cat(all_embeddings, dim=0) # [Total, 512]
    
    torch.save(final_tensor, os.path.join(OUTPUT_DIR, "wavcaps_embeddings.pt"))
    
    with open(os.path.join(OUTPUT_DIR, "wavcaps_captions.json"), 'w', encoding='utf-8') as f:
        json.dump(all_captions, f, ensure_ascii=False)
        
    print(f"✅ SUCCESS! Processed {len(all_captions)} files.")
    print(f"Output shape: {final_tensor.shape}")

if __name__ == "__main__":
    cache_datasets()
