import pandas as pd
import os
import time
import json
from pathlib import Path
from tqdm import tqdm
from google.cloud import storage
from faster_whisper import WhisperModel
from kaggle_secrets import UserSecretsClient
import threading
from concurrent.futures import ThreadPoolExecutor
import math

# ==========================================
# ⚙️ НАСТРОЙКИ (МЕНЯТЬ ЗДЕСЬ)
# ==========================================
TOTAL_HUMANS = 2 
MY_WORKER_ID = 0  # <--- ТЫ: 0, ДРУГ: 1
BUCKET_NAME = "slon_bucket2"  # <--- ИМЯ ТВОЕГО БАКЕТА

# Настройки модели
MODEL_SIZE = "large-v3"
COMPUTE_TYPE = "int8" # Для T4 лучше float16

# Пути
AUDIO_PREFIX = "audio_wav_16kHz/"
OUTPUT_FILE_NAME = f"transcriptions_part{MY_WORKER_ID}.parquet"
TEMP_DIR = Path("temp_audio")
TEMP_DIR.mkdir(exist_ok=True)

# ==========================================
# 🚫 ФИЛЬТРЫ ГАЛЛЮЦИНАЦИЙ
# ==========================================
TOXIC_PHRASES = [
    "subtitles by", "субтитры", "provided by", "сделал",
    "captioned by", "translated by", 
    "amara.org", "opensubtitles", "dimatorzok",
    "[music]", "(music)", "musikk", "[sound]", "(sound)",
    "applause", "cheering", "dramatisk", "background music",
    "copyright", "©", "🎵", "🎼", "♪"
]

CLICHE_PHRASES = [
    "thanks for watching", "спасибо за просмотр", "thanks for",
    "welcome to my channel", "subscribe", "подпишись",
    "like and subscribe", "лайк и подписка", "thank you for watching"
]

NO_SPEECH_THRESHOLD = 0.9
NO_SPEECH_TOKEN = "<NO_SPEECH>"

# Лок для принтов, чтобы потоки не мешали друг другу писать в консоль
print_lock = threading.Lock()

def safe_print(msg):
    with print_lock:
        print(msg)

def is_clean_segment(segment):
    """Проверяет текст на мусор и галлюцинации"""
    # 1. Вероятность тишины
    if segment.no_speech_prob > NO_SPEECH_THRESHOLD:
        return False

    text = segment.text.strip().lower()
    
    # 2. Слишком короткий
    if len(text) < 2:
        return False

    # 3. Токсичные фразы (удаляем всегда)
    for bad_phrase in TOXIC_PHRASES:
        if bad_phrase in text:
            return False

    # 4. Клише (удаляем, если фраза короткая)
    for cliche in CLICHE_PHRASES:
        if cliche in text:
            if len(text) < 40: return False
            
    # 5. Нет букв
    has_letters = any(c.isalpha() for c in text)
    if not has_letters: return False

    return True

# ==========================================
# ☁️ GCS HELPERS
# ==========================================
def setup_gcs_auth():
    user_secrets = UserSecretsClient()
    try:
        key_content = user_secrets.get_secret("GCP_KEY")
        key_path = "gcs_key_temp.json"
        with open(key_path, "w") as f:
            f.write(key_content)
        return key_path
    except Exception as e:
        safe_print(f"❌ Ошибка секрета GCP_KEY: {e}")
        return None

def get_gcs_client(key_path):
    return storage.Client.from_service_account_json(key_path)

def sync_from_gcs(bucket, filename):
    blob = bucket.blob(filename)
    if blob.exists():
        blob.download_to_filename(filename)
        return pd.read_parquet(filename)
    return pd.DataFrame(columns=["video_id", "text", "language", "duration"])

def save_chunk_to_gcs(df, bucket, base_filename, gpu_id):
    filename = f"{base_filename.replace('.parquet', '')}_gpu{gpu_id}.parquet"
    df.to_parquet(filename, index=False)
    blob = bucket.blob(filename)
    blob.upload_from_filename(filename)

# ==========================================
# 🧠 GPU WORKER
# ==========================================
def process_chunk(gpu_id, files, key_path):
    """Воркер для одной видеокарты (с правильным дописыванием)"""
    if not files:
        safe_print(f"💤 [GPU {gpu_id}] Нет задач.")
        return

    safe_print(f"🚀 [GPU {gpu_id}] Старт! Обработка {len(files)} файлов...")
    
    client = get_gcs_client(key_path)
    bucket = client.bucket(BUCKET_NAME)
    
    # --- 🔥 FIX #1: Загружаем предыдущий результат для этого GPU ---
    # Определяем имя файла, за который отвечает этот воркер
    chunk_filename = f"{OUTPUT_FILE_NAME.replace('.parquet', '')}_gpu{gpu_id}.parquet"
    # Скачиваем то, что уже успели сделать
    df_chunk_results = sync_from_gcs(bucket, chunk_filename)
    
    try:
        model = WhisperModel(MODEL_SIZE, device="cuda", device_index=[gpu_id], compute_type=COMPUTE_TYPE)
    except Exception as e:
        safe_print(f"❌ [GPU {gpu_id}] Ошибка загрузки модели: {e}")
        return

    # Буфер для НОВЫХ результатов
    new_rows = []
    SAVE_EVERY = 50
    
    pbar = tqdm(files, desc=f"GPU {gpu_id}", position=gpu_id, leave=True)
    
    for blob_name in pbar:
        video_id = blob_name.split('/')[-1].replace('.wav', '')
        local_wav_path = TEMP_DIR / f"{video_id}_w{MY_WORKER_ID}_g{gpu_id}.wav"
        
        try:
            blob = bucket.blob(blob_name)
            blob.download_to_filename(local_wav_path)
            
            segments, info = model.transcribe(str(local_wav_path), beam_size=5)
            
            video_duration = info.duration
            
            clean_segments_data = []
            for segment in segments:
                if segment.start > (video_duration + 1):
                    continue
                
                real_end = min(round(segment.end, 2), round(video_duration, 2))
                
                if is_clean_segment(segment):
                    clean_segments_data.append({
                        "start": round(segment.start, 2),
                        "end": real_end,
                        "text": segment.text.strip()
                    })
            
            if clean_segments_data:
                final_text_val = json.dumps(clean_segments_data, ensure_ascii=False)
            else:
                final_text_val = NO_SPEECH_TOKEN
            
            # Добавляем в буфер НОВЫХ строк
            new_rows.append({
                "video_id": video_id,
                "text": final_text_val,
                "language": info.language,
                "duration": info.duration
            })

        except Exception:
            pass
        finally:
            if local_wav_path.exists(): os.remove(local_wav_path)

        # --- 🔥 FIX #2: Логика сохранения ---
        if len(new_rows) >= SAVE_EVERY:
            # 1. Создаем DataFrame из новых строк
            df_new = pd.DataFrame(new_rows)
            # 2. Добавляем их к тому, что уже было загружено
            df_chunk_results = pd.concat([df_chunk_results, df_new], ignore_index=True)
            # 3. Сохраняем объединенный результат
            save_chunk_to_gcs(df_chunk_results, bucket, OUTPUT_FILE_NAME, gpu_id)
            # 4. Очищаем буфер
            new_rows = []

    # Финальное сохранение остатка
    if new_rows:
        df_new = pd.DataFrame(new_rows)
        df_chunk_results = pd.concat([df_chunk_results, df_new], ignore_index=True)
        save_chunk_to_gcs(df_chunk_results, bucket, OUTPUT_FILE_NAME, gpu_id)
        
    safe_print(f"🏁 [GPU {gpu_id}] Завершил работу.")
    
# ==========================================
# 🎮 MAIN CONTROLLER
# ==========================================
def run_coop_transcription():
    print(f"🔥 ЗАПУСК CO-OP (ULTIMATE) | ИГРОК {MY_WORKER_ID + 1} из {TOTAL_HUMANS} 🔥")
    
    key_path = setup_gcs_auth()
    if not key_path: return
    
    client = get_gcs_client(key_path)
    bucket = client.bucket(BUCKET_NAME)

    # 1. Проверяем, что Я уже сделал (Resume logic)
    done_ids = set()
    for g in [0, 1]:
        fname = f"{OUTPUT_FILE_NAME.replace('.parquet', '')}_gpu{g}.parquet"
        df_tmp = sync_from_gcs(bucket, fname)
        if not df_tmp.empty:
            done_ids.update(df_tmp['video_id'].astype(str))
            
    print(f"✅ Я уже сделал ранее: {len(done_ids)} видео.")

    # 2. Скачиваем список всех файлов
    print("🔎 Сканируем бакет...")
    blobs = bucket.list_blobs(prefix=AUDIO_PREFIX)
    all_wavs = sorted([b.name for b in blobs if b.name.endswith(".wav")])
    
    total_files = len(all_wavs)
    print(f"📚 Всего файлов в бакете: {total_files}")

    # 3. Делим пиццу (Global Split)
    chunk_size = math.ceil(total_files / TOTAL_HUMANS)
    start_idx = MY_WORKER_ID * chunk_size
    end_idx = min((MY_WORKER_ID + 1) * chunk_size, total_files)
    
    my_global_tasks = all_wavs[start_idx:end_idx]
    
    # 4. Фильтруем то, что уже готово
    tasks_to_do = []
    for path in my_global_tasks:
        vid = path.split('/')[-1].replace('.wav', '')
        if vid not in done_ids:
            tasks_to_do.append(path)

    print(f"🔥 Мне осталось сделать: {len(tasks_to_do)}")
    
    if not tasks_to_do:
        print("🎉 Моя часть работы выполнена! Можно отдыхать.")
        return

    # 5. Делим на 2 GPU (Local Split)
    mid = len(tasks_to_do) // 2
    tasks_gpu0 = tasks_to_do[:mid]
    tasks_gpu1 = tasks_to_do[mid:]

    # 6. Запуск потоков
    with ThreadPoolExecutor(max_workers=2) as executor:
        f0 = executor.submit(process_chunk, 0, tasks_gpu0, key_path)
        f1 = executor.submit(process_chunk, 1, tasks_gpu1, key_path)
        f0.result()
        f1.result()

    print("\n🎉🎉🎉 СЕССИЯ ЗАВЕРШЕНА!")

if __name__ == "__main__":
    run_coop_transcription()
