import pandas as pd
import os
import time
from pathlib import Path
from tqdm import tqdm
from google.cloud import storage
from faster_whisper import WhisperModel
from kaggle_secrets import UserSecretsClient
import threading
from concurrent.futures import ThreadPoolExecutor
import math

# --- ⚙️ НАСТРОЙКИ ДЛЯ КОМАНДНОЙ РАБОТЫ ⚙️ ---
# 1. Сколько человек участвует? (Если ты и друг, ставь 2)
TOTAL_HUMANS = 2 

# 2. Кто запускает этот скрипт? (Меняйте это число!)
# Ты ставишь 0
# Друг ставит 1
MY_WORKER_ID = 0  

# 3. Уникальное имя файла для каждого (чтобы не перезатереть друг друга)
# Например: "transcriptions_part0.parquet" и "transcriptions_part1.parquet"
OUTPUT_FILE_NAME = f"transcriptions_part{MY_WORKER_ID}.parquet"

# --- ОСТАЛЬНАЯ КОНФИГУРАЦИЯ ---
BUCKET_NAME = "slon_bucket2"
AUDIO_PREFIX = "audio_wav_16kHz/"

# Настройки модели
MODEL_SIZE = "large-v3"
COMPUTE_TYPE = "int8"

TEMP_DIR = Path("temp_audio")
TEMP_DIR.mkdir(exist_ok=True)

# Глобальный лок для принтов
print_lock = threading.Lock()

def safe_print(msg):
    with print_lock:
        print(msg)

def setup_gcs_auth():
    user_secrets = UserSecretsClient()
    try:
        key_content = user_secrets.get_secret("GCP_KEY")
        key_path = "gcs_key_temp.json"
        with open(key_path, "w") as f:
            f.write(key_content)
        return key_path
    except Exception as e:
        safe_print(f"❌ Ошибка секрета: {e}")
        return None

def get_gcs_client(key_path):
    return storage.Client.from_service_account_json(key_path)

def sync_from_gcs(bucket, filename):
    blob = bucket.blob(filename)
    if blob.exists():
        blob.download_to_filename(filename)
        return pd.read_parquet(filename)
    return pd.DataFrame(columns=["video_id", "text", "language", "duration"])

def save_chunk_to_gcs(df, bucket, base_filename, gpu_id):
    # Добавляем суффикс GPU к имени файла воркера
    # Пример: transcriptions_part0_gpu1.parquet
    filename = f"{base_filename.replace('.parquet', '')}_gpu{gpu_id}.parquet"
    df.to_parquet(filename, index=False)
    blob = bucket.blob(filename)
    blob.upload_from_filename(filename)

def process_chunk(gpu_id, files, key_path):
    """Воркер для одной видеокарты"""
    if not files:
        safe_print(f"💤 [GPU {gpu_id}] Нет задач.")
        return

    safe_print(f"🚀 [GPU {gpu_id}] Старт! Обработка {len(files)} файлов...")
    
    client = get_gcs_client(key_path)
    bucket = client.bucket(BUCKET_NAME)
    
    try:
        model = WhisperModel(MODEL_SIZE, device="cuda", device_index=[gpu_id], compute_type=COMPUTE_TYPE)
    except Exception as e:
        safe_print(f"❌ [GPU {gpu_id}] Ошибка загрузки модели: {e}")
        return

    local_results = []
    SAVE_EVERY = 50
    
    pbar = tqdm(files, desc=f"GPU {gpu_id}", position=gpu_id, leave=True)
    
    for blob_name in pbar:
        video_id = blob_name.split('/')[-1].replace('.wav', '')
        local_wav_path = TEMP_DIR / f"{video_id}_w{MY_WORKER_ID}_g{gpu_id}.wav"
        
        try:
            blob = bucket.blob(blob_name)
            blob.download_to_filename(local_wav_path)
            
            segments, info = model.transcribe(str(local_wav_path), beam_size=5)
            full_text = " ".join([segment.text for segment in segments]).strip()
            
            local_results.append({
                "video_id": video_id,
                "text": full_text,
                "language": info.language,
                "duration": info.duration
            })

        except Exception:
            pass 
        finally:
            if local_wav_path.exists(): os.remove(local_wav_path)

        if len(local_results) % SAVE_EVERY == 0:
            df_chunk = pd.DataFrame(local_results)
            save_chunk_to_gcs(df_chunk, bucket, OUTPUT_FILE_NAME, gpu_id)

    if local_results:
        df_chunk = pd.DataFrame(local_results)
        save_chunk_to_gcs(df_chunk, bucket, OUTPUT_FILE_NAME, gpu_id)
        
    safe_print(f"🏁 [GPU {gpu_id}] Завершил работу.")

def run_coop_transcription():
    print(f"🔥 ЗАПУСК CO-OP РЕЖИМА | ИГРОК {MY_WORKER_ID + 1} из {TOTAL_HUMANS} 🔥")
    print(f"📂 Мой файл результата: {OUTPUT_FILE_NAME}")
    
    key_path = setup_gcs_auth()
    if not key_path: return
    
    client = get_gcs_client(key_path)
    bucket = client.bucket(BUCKET_NAME)

    # 1. Проверяем, что Я уже сделал (мои gpu0 и gpu1 файлы)
    done_ids = set()
    for g in [0, 1]:
        fname = f"{OUTPUT_FILE_NAME.replace('.parquet', '')}_gpu{g}.parquet"
        df_tmp = sync_from_gcs(bucket, fname)
        if not df_tmp.empty:
            done_ids.update(df_tmp['video_id'].astype(str))
            
    print(f"✅ Я уже сделал ранее: {len(done_ids)} видео.")

    # 2. Скачиваем ПОЛНЫЙ список файлов
    print("🔎 Сканируем бакет...")
    blobs = bucket.list_blobs(prefix=AUDIO_PREFIX)
    all_wavs = sorted([b.name for b in blobs if b.name.endswith(".wav")])
    
    total_files = len(all_wavs)
    print(f"📚 Всего файлов в бакете: {total_files}")

    # 3. ДЕЛИМ ПИЦЦУ (Global Split)
    # Вычисляем мою долю
    chunk_size = math.ceil(total_files / TOTAL_HUMANS)
    start_idx = MY_WORKER_ID * chunk_size
    end_idx = min((MY_WORKER_ID + 1) * chunk_size, total_files)
    
    my_global_tasks = all_wavs[start_idx:end_idx]
    
    print("-" * 40)
    print(f"🍕 Мой кусок пиццы: с {start_idx} по {end_idx}")
    print(f"📋 Всего мне поручено: {len(my_global_tasks)} файлов")
    print("-" * 40)

    # 4. Фильтруем то, что я уже сделал
    tasks_to_do = []
    for path in my_global_tasks:
        vid = path.split('/')[-1].replace('.wav', '')
        if vid not in done_ids:
            tasks_to_do.append(path)

    print(f"🔥 Мне осталось сделать: {len(tasks_to_do)}")
    
    if not tasks_to_do:
        print("🎉 Моя часть работы выполнена! Можно пить чай.")
        return

    # 5. Делим на мои 2 GPU (Local Split)
    mid = len(tasks_to_do) // 2
    tasks_gpu0 = tasks_to_do[:mid]
    tasks_gpu1 = tasks_to_do[mid:]

    # 6. Погнали
    with ThreadPoolExecutor(max_workers=2) as executor:
        f0 = executor.submit(process_chunk, 0, tasks_gpu0, key_path)
        f1 = executor.submit(process_chunk, 1, tasks_gpu1, key_path)
        f0.result()
        f1.result()

    print("\n🎉🎉🎉 СЕССИЯ ЗАВЕРШЕНА! Не забудь объединить файлы в конце.")

if __name__ == "__main__":
    run_coop_transcription()
